CREATE TABLE L8CUSTOMER(
CID NUMBER(3) PRIMARY KEY,
CNAME VARCHAR(30));

CREATE TABLE L8BRANCH(
BCODE NUMBER(3) PRIMARY KEY,
BNAME VARCHAR(30));

CREATE TABLE L8ACCOUNT(
ANO NUMBER(3) PRIMARY KEY,
ATYPE VARCHAR(1) CHECK(ATYPE IN ('S','C')),
BALANCE NUMBER(8) DEFAULT 0,
CID NUMBER(3),
BCODE NUMBER(3),
FOREIGN KEY (CID) REFERENCES L8CUSTOMER(CID),
FOREIGN KEY (BCODE) REFERENCES L8BRANCH(BCODE)
);

CREATE TABLE L8TRANSACTION(
TID NUMBER(3) PRIMARY KEY,
ANO NUMBER(3),
TTYPE VARCHAR(1) CHECK(TTYPE IN ('D','W')),
TDATE DATE,
TAMOUNT NUMBER(8),
FOREIGN KEY (ANO) REFERENCES L8ACCOUNT(ANO));

CREATE OR REPLACE TRIGGER TRG_RESTRNX BEFORE INSERT ON L8TRANSACTION 
FOR EACH ROW
DECLARE
 TCO NUMBER;
BEGIN
SELECT COUNT(*) INTO TCO FROM L8TRANSACTION WHERE ANO=:NEW.ANO AND TDATE=:NEW.TDATE;
IF TCO>2 THEN 
RAISE_APPLICATION_ERROR(-20001,'ENROLL FAILED');
END IF;
END;
/

(I)
CREATE OR REPLACE TRIGGER TRG_DEPO AFTER INSERT ON L8TRANSACTION 
FOR EACH ROW 
BEGIN 
IF :NEW.TTYPE='D' THEN
UPDATE L8ACCOUNT SET BALANCE= BALANCE + :NEW.TAMOUNT WHERE ANO=:NEW.ANO;
ELSE
UPDATE L8ACCOUNT SET BALANCE= BALANCE - :NEW.TAMOUNT WHERE ANO=:NEW.ANO;
END IF;
END;
/

CREATE OR REPLACE TRIGGER TRG_WITH BEFORE INSERT ON L8TRANSACTION
FOR EACH ROW 
DECLARE 
  TYP VARCHAR2(1);
  BAL NUMBER;
  MIN_BAL NUMBER; 
BEGIN
  IF :NEW.TTYPE = 'W' THEN 
    SELECT BALANCE, ATYPE INTO BAL, TYP FROM L8ACCOUNT WHERE ANO = :NEW.ANO;
    
    IF TYP = 'S' THEN 
      MIN_BAL := 2000;
    ELSE 
      MIN_BAL := 5000;
    END IF;

    IF (BAL - :NEW.TAMOUNT) < MIN_BAL THEN
      RAISE_APPLICATION_ERROR(-20001, 'INSUFFICIENT FUND');
    END IF;
  END IF;
END;
/


(J)
CREATE OR REPLACE TRIGGER TRG_DAYLIM BEFORE INSERT ON L8TRANSACTION
FOR EACH ROW 
DECLARE
 TOTAMT NUMBER;
BEGIN
SELECT SUM(TAMOUNT) INTO TOTAMT FROM L8TRANSACTION GROUP BY ANO HAVING TTYPE='W' AND TDATE=:NEW.TDATE;
IF TOTAMT>30000 THEN 
RAISE_APPLICATION_ERROR(-20001,'MAX LIMIT REACHED');
END IF;
END;
/

CREATE OR REPLACE TRIGGER TRG_DAYLIM
FOR INSERT ON L8TRANSACTION
COMPOUND TRIGGER
  TOTAMT NUMBER := 0;

  AFTER EACH ROW IS
  BEGIN
    IF :NEW.TTYPE = 'W' THEN
      SELECT NVL(SUM(TAMOUNT), 0) INTO TOTAMT 
      FROM L8TRANSACTION 
      WHERE ANO = :NEW.ANO 
        AND TTYPE = 'W' 
        AND TDATE = :NEW.TDATE;

      IF (TOTAMT + :NEW.TAMOUNT) > 30000 THEN
        RAISE_APPLICATION_ERROR(-20001, 'Daily withdrawal limit of 30,000 reached for Account');
      END IF;
    END IF;
  END AFTER EACH ROW;
END;
/

CREATE OR REPLACE PROCEDURE PRC_UPDATE_BALANCE (
    p_type IN VARCHAR2, 
    p_amt  IN NUMBER, 
    p_ano  IN NUMBER
) IS
BEGIN
    IF p_type = 'D' THEN
        UPDATE L8ACCOUNT SET BALANCE = BALANCE + p_amt WHERE ANO = p_ano;
    ELSIF p_type = 'W' THEN
        UPDATE L8ACCOUNT SET BALANCE = BALANCE - p_amt WHERE ANO = p_ano;
    END IF;
END;
/
